# ObjectHandler — builds the static libraries consumed by the XLL.
#
# This directory can be configured in TWO ways:
#
#   * As part of the top-level build (add_subdirectory from the parent).
#
#   * Standalone — `cmake -S ObjectHandler -B <build>` — to build just the
#     ObjectHandler libraries (and, with -DBUILD_DOCS=ON, its documentation)
#     with no reference to any downstream project.  The only external piece it
#     relies on is the generic build settings in
#     gensrc/cmake/commonSettings.cmake (gensrc is ObjectHandler's upstream
#     dependency, so referring to it does not violate the project hierarchy).
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.15)
    # Required for CMAKE_MSVC_RUNTIME_LIBRARY (CMP0091) — must precede project().
    cmake_policy(SET CMP0091 NEW)
    project(ObjectHandler CXX)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../gensrc/cmake/commonSettings.cmake")
endif()

# -----------------------------------------------------------------------
# gensrc — optional code-generation step (the "Full" build), standalone only.
#
# Several of the ohxllib sources (ohxl/functions/*.cpp, ohxl/register/*.cpp)
# are autogenerated by gensrc and are not committed to the repository, so on a
# pristine clone they do not exist yet.  CMake checks that the sources listed
# in add_library() exist when the target is *created*, i.e. at configure time;
# a build-time-only target would run too late and the first configure of
# ohxllib would fail with "Cannot find source file".  When RUN_GENSRC is ON we
# therefore run gensrc here, DURING configuration, so the generated sources
# exist before the ohxllib target below is created.  (A build-time ohgensrc
# target is also defined further down, after the library targets, so that an
# incremental build refreshes the generated sources before compiling.)
#
# This block is guarded by the standalone check, so in the integrated (root)
# build it is skipped — the root CMakeLists.txt owns the gensrc wiring.  gensrc
# is ObjectHandler's upstream dependency, so invoking ../gensrc/gensrc.py does
# not reference any downstream project, respecting the project hierarchy.
# -----------------------------------------------------------------------
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    option(RUN_GENSRC "Run gensrc to regenerate addin source before building" OFF)

    if(RUN_GENSRC)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)

        # The ObjectHandler gensrc command (Excel, Doxygen, Loops, ValueObjects),
        # run from ObjectHandler/gensrc with --oh_dir=.. pointing back at this
        # directory.  Mirrors the ohgensrc step of the integrated build.
        set(_ohgensrc_cmd ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/../gensrc/gensrc.py" -xdlv --oh_dir=..)
        set(_ohgensrc_wd  "${CMAKE_CURRENT_SOURCE_DIR}/gensrc")

        # Configure-time generation, recorded with a stamp file so it only runs
        # once per build tree.  Delete the stamp (or configure into a clean build
        # directory) to force regeneration.
        set(_ohgensrc_stamp "${CMAKE_CURRENT_BINARY_DIR}/ohgensrc.configured.stamp")
        if(NOT EXISTS "${_ohgensrc_stamp}")
            message(STATUS "RUN_GENSRC: running gensrc for ObjectHandler (configure-time)")
            execute_process(
                COMMAND ${_ohgensrc_cmd}
                WORKING_DIRECTORY ${_ohgensrc_wd}
                RESULT_VARIABLE _ohgensrc_rc
                OUTPUT_VARIABLE _ohgensrc_out
                ERROR_VARIABLE  _ohgensrc_out)
            if(NOT _ohgensrc_rc EQUAL 0)
                message(FATAL_ERROR "gensrc (ObjectHandler) failed with code ${_ohgensrc_rc}:\n${_ohgensrc_out}")
            endif()
            file(WRITE "${_ohgensrc_stamp}" "gensrc ran at configure time\n")
        endif()
    endif()
endif()

# Ensure Boost imported targets are available (find_package is idempotent).
# Static Boost (and static runtime) is only forced for the MSVC/Windows build;
# on Linux we use the platform's default Boost libraries.
if(MSVC)
    if(NOT MSVC_LINK_DYNAMIC_RUNTIME)
        set(Boost_USE_STATIC_RUNTIME ON)
    endif()
    set(Boost_USE_STATIC_LIBS ON)
endif()
find_package(Boost 1.58.0 REQUIRED COMPONENTS filesystem serialization)


#   xlsdk          : Excel SDK (framewrk.cpp + xlcall32.lib)   [Windows only]
#   ohlib          : ObjectHandler core (oh/*.cpp)
#   ohxllib        : ObjectHandler Excel layer (ohxl/**/*.cpp) [Windows only]

# -----------------------------------------------------------------------
# xlsdk  (Windows/Excel only)
# -----------------------------------------------------------------------
if(WIN32)
add_library(xlsdk STATIC
    xlsdk/framewrk.cpp
)
# Namespaced alias so consumers (e.g. ohxllib, QuantLibXL) can link the same
# ObjectHandler::xlsdk name whether ObjectHandler is added in-tree or imported
# via find_package(ObjectHandler CONFIG) from a standalone build.
add_library(ObjectHandler::xlsdk ALIAS xlsdk)

target_include_directories(xlsdk PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
)

# Suppress the xlsdk/auto_link.hpp #pragma comment(lib, ...) directive; cmake
# manages linking.  PUBLIC so every consumer of xlsdk inherits it too.
target_compile_definitions(xlsdk PUBLIC XLSDK_NO_AUTO_LINK)

# Link against the pre-built xlcall32.lib that ships with the SDK.
# x64 copy lives in xlsdk/x64/; Win32 copy is xlsdk/xlcall32.lib.
target_link_libraries(xlsdk PUBLIC
    $<IF:$<EQUAL:${CMAKE_SIZEOF_VOID_P},8>,
        ${CMAKE_CURRENT_SOURCE_DIR}/xlsdk/x64/XLCALL32.LIB,
        ${CMAKE_CURRENT_SOURCE_DIR}/xlsdk/xlcall32.lib>
)
endif()

# -----------------------------------------------------------------------
# gensrc — build-time regeneration target (the "Full" build), standalone only.
#
# Companion to the configure-time generation near the top of this file: the
# ohgensrc custom target re-runs gensrc on an incremental build so that edits
# to the gensrc metadata/templates refresh the generated ohxl sources before
# compilation.  ohlib is wired to depend on it (and ohxllib links ohlib), so
# the generated sources are refreshed before either library is compiled —
# mirroring the ohgensrc dependency in the integrated build.  Skipped in the
# integrated build, where the root CMakeLists.txt owns this wiring.
# -----------------------------------------------------------------------
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND RUN_GENSRC)
    add_custom_target(ohgensrc
        COMMAND ${_ohgensrc_cmd}
        WORKING_DIRECTORY ${_ohgensrc_wd}
        COMMENT "Running gensrc for ObjectHandler"
        VERBATIM)
    add_dependencies(ohlib ohgensrc)
endif()

# -----------------------------------------------------------------------
# Build-tree package export (standalone only).
#
# When ObjectHandler is configured on its own, write a CMake package config
# into the build tree so a downstream *standalone* build (QuantLibAddin,
# QuantLibXL) can consume the libraries we just built with
#     find_package(ObjectHandler CONFIG)
# pointed at this build directory — with no install step.  The exported
# imported targets are ObjectHandler::ohlib (and, on Windows,
# ObjectHandler::xlsdk / ObjectHandler::ohxllib).
#
# Skipped in the integrated (root) build, where downstream projects link the
# in-tree ObjectHandler::* aliases directly.  This block references no
# downstream project, so it respects the project dependency hierarchy.
# -----------------------------------------------------------------------
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(_oh_export_targets ohlib)
    if(WIN32)
        list(APPEND _oh_export_targets xlsdk ohxllib)
    endif()

    export(TARGETS ${_oh_export_targets}
        NAMESPACE ObjectHandler::
        FILE "${CMAKE_CURRENT_BINARY_DIR}/cmake/ObjectHandlerTargets.cmake")

    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/cmake/ObjectHandlerConfig.cmake.in"
        "${CMAKE_CURRENT_BINARY_DIR}/cmake/ObjectHandlerConfig.cmake"
        COPYONLY)
endif()
