cmake_minimum_required(VERSION 3.15)

# Required for CMAKE_MSVC_RUNTIME_LIBRARY (CMP0091)
cmake_policy(SET CMP0091 NEW)

project(QuantLibAddin CXX)

include(cmake/commonSettings.cmake)

# -----------------------------------------------------------------------
# gensrc — optional code-generation step (the "Full" build).
# Defines the QLA_RUN_GENSRC option and the ohgensrc / qlgensrc targets.
# The dependency wiring onto the library targets (ohlib, QuantLibObjects)
# is done further below, once those targets have been created.
# -----------------------------------------------------------------------
add_subdirectory(gensrc)

# -----------------------------------------------------------------------
# QuantLib (adds target ql_library, aliased as QuantLib::QuantLib)
# -----------------------------------------------------------------------
set(QL_BUILD_EXAMPLES   OFF CACHE BOOL "" FORCE)
set(QL_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
# QL_TAGGED_LAYOUT is ON by default when MSVC is used, which sets
# CMAKE_RELEASE_POSTFIX / CMAKE_DEBUG_POSTFIX globally.  Those global
# postfixes would affect every cmake target in the build, not just
# ql_library.  We therefore turn the flag OFF here so that only
# ql_library gets its postfix set (we do that explicitly below).
set(QL_TAGGED_LAYOUT OFF CACHE BOOL "" FORCE)
add_subdirectory(QuantLib)

# Apply tagged layout postfixes to ql_library only, mirroring what
# QL_TAGGED_LAYOUT would have done, so the library filename matches
# the filenames expected by the hand-maintained VS build.
if(MSVC)
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(_ql_arch_tag "-x64")
    else()
        set(_ql_arch_tag "")
    endif()
    if(MSVC_LINK_DYNAMIC_RUNTIME)
        set(_ql_debug_postfix   "${_ql_arch_tag}-mt-gd")
        set(_ql_release_postfix "${_ql_arch_tag}-mt")
    else()
        set(_ql_debug_postfix   "${_ql_arch_tag}-mt-sgd")
        set(_ql_release_postfix "${_ql_arch_tag}-mt-s")
    endif()
    set_target_properties(ql_library PROPERTIES
        DEBUG_POSTFIX          "${_ql_debug_postfix}"
        RELEASE_POSTFIX        "${_ql_release_postfix}"
        RELWITHDEBINFO_POSTFIX "${_ql_release_postfix}"
        MINSIZEREL_POSTFIX     "${_ql_release_postfix}"
    )
endif()

# -----------------------------------------------------------------------
# ObjectHandler static libs (ohlib + ohxllib) and xlsdk
# -----------------------------------------------------------------------
add_subdirectory(ObjectHandler)

# -----------------------------------------------------------------------
# QuantLibObjects static lib (QuantLibAddin qlo wrapper)
# -----------------------------------------------------------------------
add_subdirectory(QuantLibAddin)

# -----------------------------------------------------------------------
# QuantLibXL — the Excel XLL (Windows only)
# -----------------------------------------------------------------------
if(WIN32)
    add_subdirectory(QuantLibXL)
endif()

# -----------------------------------------------------------------------
# When the "Full" build is enabled, make the addin libraries depend on the
# gensrc targets so the autogenerated sources are refreshed before compiling.
# -----------------------------------------------------------------------
if(QLA_RUN_GENSRC)
    add_dependencies(ohlib ohgensrc)
    add_dependencies(QuantLibObjects qlgensrc)
endif()

# -----------------------------------------------------------------------
# Documentation — adds per-project doc targets (gensrc-docs, ObjectHandler-docs,
# QuantLibAddin-docs, QuantLibXL-docs) and an aggregate `docs` target that
# build the HTML documentation with Doxygen, mirroring the legacy automake
# "make docs".  Off by default; enable with -DQLA_BUILD_DOCS=ON.
#
# Each project keeps its own documentation build in <project>/Docs/CMakeLists.txt
# so that, like the rest of the repository, a project's docs are part of that
# standalone project and can be built on their own, e.g.:
#     cmake -S ObjectHandler/Docs -B <build> && cmake --build <build> --target docs
#
# Building the docs needs only Python 3, Doxygen and (optionally) Graphviz dot —
# not a C++ compiler or Boost — so the documentation can be built even on a
# machine that cannot compile the libraries.
#
# QuantLibAddin/Docs is added before QuantLibXL/Docs because the QuantLibXL docs
# reuse the QuantLibAddin-docs-gensrc target defined by QuantLibAddin/Docs.
# -----------------------------------------------------------------------
option(QLA_BUILD_DOCS "Add targets to build the HTML documentation with Doxygen" OFF)
if(QLA_BUILD_DOCS)
    add_subdirectory(gensrc/Docs)
    add_subdirectory(ObjectHandler/Docs)
    add_subdirectory(QuantLibAddin/Docs)
    add_subdirectory(QuantLibXL/Docs)

    # Aggregate target: build every project's documentation with one command.
    add_custom_target(docs
        COMMENT "Building all HTML documentation (gensrc, ObjectHandler, QuantLibAddin, QuantLibXL)")
    add_dependencies(docs
        gensrc-docs
        ObjectHandler-docs
        QuantLibAddin-docs
        QuantLibXL-docs)
endif()
